let workerInside = false;
let entryStartTime = null;
let timerInterval = null;

function checkEntry() {

    const workerId =
        document.getElementById("workerId").value.trim();

    if (workerId === "") {
        alert("Please enter Worker ID");
        return;
    }

    const status =
        document.getElementById("status").innerText;

    const result =
        document.getElementById("entryResult");

    if (status === "SAFE") {

        workerInside = true;

        entryStartTime = new Date();

        document.getElementById("entryTime").innerText =
            entryStartTime.toLocaleTimeString();

        result.innerText =
            "ENTRY APPROVED for " + workerId;

        result.style.color = "#15803d";

        startTimer();

        addEvent(
            "Worker Entry Approved",
            "Worker " + workerId + " entered the sewer area",
            "safe"
        );

    } else {

        result.innerText =
            "ENTRY DENIED - DO NOT ENTER";

        result.style.color = "#dc2626";

        addEvent(
            "Entry Denied",
            "Worker " + workerId + " was denied entry due to unsafe conditions",
            "danger"
        );
    }
}

function startTimer() {

    clearInterval(timerInterval);

    timerInterval = setInterval(() => {

        if (!workerInside || !entryStartTime) {
            return;
        }

        const now = new Date();

        const difference =
            Math.floor(
                (now - entryStartTime) / 1000
            );

        const hours =
            Math.floor(difference / 3600);

        const minutes =
            Math.floor(
                (difference % 3600) / 60
            );

        const seconds =
            difference % 60;

        document.getElementById("timer").innerText =
            String(hours).padStart(2, "0") + ":" +
            String(minutes).padStart(2, "0") + ":" +
            String(seconds).padStart(2, "0");

    }, 1000);
}

function workerExit() {

    if (!workerInside) {

        alert("No worker is currently inside");

        return;
    }

    const workerId =
        document.getElementById("workerId").value;

    workerInside = false;

    clearInterval(timerInterval);

    document.getElementById("timer").innerText =
        "00:00:00";

    document.getElementById("entryResult").innerText =
        "Worker exited safely";

    document.getElementById("entryResult").style.color =
        "#15803d";

    addEvent(
        "Worker Exit",
        "Worker " + workerId + " exited safely",
        "safe"
    );
}

function addEvent(title, message, type) {

    const events =
        document.getElementById("events");

    const event =
        document.createElement("div");

    event.className = "event";

    let dotClass = "safe-dot";

    if (type === "danger") {
        dotClass = "danger-dot";
    }

    event.innerHTML = `
        <div class="event-dot ${dotClass}"></div>

        <div>
            <strong>${title}</strong>
            <p>${message}</p>
            <small>${new Date().toLocaleTimeString()}</small>
        </div>
    `;

    events.prepend(event);
}

function simulateDanger() {

    const status =
        document.getElementById("status");

    status.innerText = "DANGER";

    status.className = "status danger";

    document.getElementById("statusMessage").innerText =
        "Danger detected! Entry is not permitted.";

    document.getElementById("entryPermission").innerText =
        "DENIED";

    document.getElementById("entryPermission").style.color =
        "#dc2626";

    document.getElementById("blower").innerText =
        "ON";

    document.getElementById("blowerStatus").innerText =
        "ON";

    document.getElementById("alarmStatus").innerText =
        "ON";

    document.getElementById("supervisorStatus").innerText =
        "ALERT";

    document.getElementById("responseStatus").innerText =
        "Danger Detected";

    addEvent(
        "Hazard Detected",
        "Unsafe environmental condition detected",
        "danger"
    );
}
